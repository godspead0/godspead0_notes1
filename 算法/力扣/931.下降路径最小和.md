// @lc code=start
class Solution {
public:
    int minFallingPathSum(vector<vector<int>>& matrix) {
        int n = matrix.size();
        vector<vector<int>> dp(n+1, vector<int>(n+2 ,1e9));

        for(int j = 1;j <= n;j ++){
            dp[n][j] = matrix[n - 1][j - 1];
        }

        for(int i = n - 1;i >= 1;i --){
            for(int j = n;j >= 1;j --){
                dp[i][j] = min(dp[i + 1][j - 1],min(dp[i + 1][j],dp[i + 1][j + 1])) + matrix[i - 1][j - 1];
            }
        }
        int ans = 1e9;
        for(int j = 1;j <= n;j ++){
            ans = min(ans, dp[1][j]);
        }
        return ans;
    }
};
// @lc code=end
