# [215] 数组中的第K个最大元素

/*
 * @lc app=leetcode.cn id=215 lang=cpp

 * [215] 数组中的第K个最大元素

// @lc code=start
class Solution {
public: 
    int quick(vector<int> &nums, int l, int r, int k){
        if(r == l) return nums[k];
        int flag = nums[l];
        int i = l - 1, j = r + 1;
        while(i < j){
            do i++; while(nums[i] < flag);
            do j--; while(nums[j] > flag);
            if(i < j) swap(nums[i], nums[j]);
        }
        if(k <= j) return quick(nums, l, j, k);
        else return quick(nums, j + 1, r, k);
    }
    int findKthLargest(vector<int>& nums, int k) {
        return quick(nums, 0, nums.size() - 1, nums.size() - k);
    }
};

// @lc code=end
