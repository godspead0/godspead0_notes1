// @lc code=start
class Solution {
public:
    int tribonacci(int n) {
        int dp[40] = {0, 1, 1};
        for (int i = 3; i <= n; i++) {
            dp[i] = dp[i - 1] + dp[i - 2] + dp[i - 3];
        }
        return dp[n];
    }
};
// @lc code=end
