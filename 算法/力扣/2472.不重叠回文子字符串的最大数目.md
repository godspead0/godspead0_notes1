// @lc code=start
class Solution {
public:
    int maxPalindromes(string s, int k) {
        int n = s.size();
        int ans = 0, start = 0;
        auto cmp = [&](int l, int r){
            while(l < r){
                if(s[l++] != s[r--]) return false;
            }
            return true;
        };
        for(int r = k - 1;r < n;r ++){
            int l = r - k + 1;
            if(l >= start && cmp(l, r)){
                ans++;
                start = r + 1;
                continue;
            }
            l = r - k;
            if(l >= start && cmp(l, r)){
                ans++;
                start = r + 1;
            }
        }
        return ans;
    }
};
// @lc code=end
