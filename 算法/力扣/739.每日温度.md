// @lc code=start
class Solution {
public:
    vector<int> dailyTemperatures(vector<int>& temperatures) {
        stack<int> t;
        vector<int> v(temperatures.size());
        for(int i = 0;i < temperatures.size();i ++){
            while(!t.empty() && temperatures[i] > temperatures[t.top()]){
                v[t.top()] = i - t.top();
                t.pop();
            }
            t.push(i);
        }
        v[v.size() - 1] = 0;
        return v;
    }
};
