class Solution {
public:
    bool canJump(vector<int>& nums) {
        int n = nums.size();
        vector<int> dp(n);
        dp[0] = nums[0];
        if(nums[0] >= n - 1) return true;
        for(int i = 1;i < n;i ++){
            if(dp[i - 1] < i) return false;
            dp[i] = max(dp[i - 1], nums[i] + i);
            if(dp[i] >= n - 1) return true;
        }
        return dp[n - 1] >= n - 1;
    }
};
