# 合格的机器

```cpp
https://ac.nowcoder.com/acm/contest/122727/A

#include<bits/stdc++.h>
using namespace std;
int main(){
    int n = 0;
    cin>>n;
    int sum = 0;
    for(int i = 0;i < n;i ++){
        int m = 0;
        cin>>m;
        if(m==0)sum--;
        else if(m>1)sum+=(m-1);
    }
    if(sum<=n)cout<<sum;
    else{
        if((sum-n)%2==1)cout<<n-1;
        else cout<<n;
    }
    return 0;
}
```
