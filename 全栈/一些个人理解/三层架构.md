```plaintext
vue-project/
├── src/                     # 源码目录
│   ├── api/                 # 接口请求封装：统一管理后端接口
│   │   ├── user.js          # 用户模块接口（如登录、查询）
│   │   ├── order.js         # 订单模块接口
│   │   └── request.js       # axios 封装（拦截器、请求头、响应处理）
│   ├── assets/              # 静态资源：图片、样式、字体
│   │   ├── images/          # 图片资源（如 logo.png）
│   │   └── styles/          # 全局样式（如 reset.css、variable.scss）
│   ├── components/          # 公共组件：可复用的 UI 组件
│   │   ├── CommonButton.vue # 自定义按钮组件
│   │   ├── Pagination.vue   # 分页组件
│   │   └── layout/          # 布局组件（如 Header.vue、Sidebar.vue）
│   ├── router/              # 路由配置：页面跳转规则
│   │   └── index.js         # 路由定义（懒加载、路由守卫）
│   ├── store/               # 状态管理：Vuex/Pinia（Vue3 推荐 Pinia）
│   │   ├── modules/         # 模块化状态（如 userStore.js）
│   │   └── index.js         # 状态管理入口
│   ├── views/               # 页面组件：对应路由的页面
│   │   ├── login/           # 登录页面
│   │   │   └── Login.vue
│   │   ├── user/            # 用户模块页面
│   │   │   ├── UserList.vue # 用户列表
│   │   │   └── UserDetail.vue # 用户详情
│   │   └── order/           # 订单模块页面
│   ├── utils/               # 前端工具类（如日期格式化、权限判断）
│   │   └── auth.js          # 登录状态管理（localStorage/cookie）
│   ├── App.vue              # 根组件：页面入口
│   └── main.js              # 入口文件：创建 Vue 实例、挂载插件
├── public/                  # 静态资源（不经过打包，直接访问）
│   └── favicon.ico          # 网站图标
├── package.json             # 项目依赖（vue、axios、element-plus 等）
└── vite.config.js           # Vite 配置（代理、插件、别名）
```
